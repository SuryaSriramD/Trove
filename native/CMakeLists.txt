cmake_minimum_required(VERSION 3.21)
project(trove_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/workers")
# Configure llama and whisper in separate build trees: their ggml targets differ.
if(TROVE_RUNTIME STREQUAL "llama")
  add_subdirectory("${TROVE_RUNTIME_SOURCE}" runtime)
  add_executable(llama-worker llama-worker.cpp)
  target_link_libraries(llama-worker PRIVATE llama)
  add_executable(vision-worker vision-worker.cpp)
  target_link_libraries(vision-worker PRIVATE mtmd llama)
  set(workers llama-worker vision-worker)
elseif(TROVE_RUNTIME STREQUAL "whisper")
  add_subdirectory("${TROVE_RUNTIME_SOURCE}" runtime)
  add_executable(whisper-worker whisper-worker.cpp)
  target_link_libraries(whisper-worker PRIVATE whisper)
  set(workers whisper-worker)
else()
  message(FATAL_ERROR "TROVE_RUNTIME must be llama or whisper")
endif()
find_package(Threads REQUIRED)
if(NOT WIN32 AND NOT APPLE)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
foreach(worker IN LISTS workers)
  target_include_directories(${worker} PRIVATE "${TROVE_JSON_INCLUDE}")
  target_link_libraries(${worker} PRIVATE Threads::Threads)
  if(WIN32)
    target_compile_definitions(${worker} PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(${worker} PRIVATE bcrypt)
  elseif(NOT APPLE)
    target_link_libraries(${worker} PRIVATE OpenSSL::Crypto)
  endif()
endforeach()
